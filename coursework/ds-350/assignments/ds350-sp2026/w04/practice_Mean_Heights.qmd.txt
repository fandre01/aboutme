```{r}
# Let's clean up the column names and pivot the years
country_data <- heights_raw %>% 
  rename(country = `Continent, Region, Country`) %>% 
  pivot_longer(
    cols = `1800`:`2011`,
    names_to = "year",
    values_to = "height.cm"
  )
country_data
```

```{r}
tidy_heights <- country_data %>% 
  # Drop rows where height is missing
  filter(!is.na(height.cm)) %>% 
  # Convert year to a number so we can calculate decades
  mutate(
    year = as.numeric(year),
    year_decade = (year %/% 10) * 10
  ) %>% 
  # Group by country and decade to find the mean height requested
  group_by(Code, country, year_decade) %>% 
  summarize(height.cm = mean(height.cm, na.rm = TRUE), .groups = "drop") %>%

  select(Code, country, year_decade, height.cm) %>%
  arrange(country)
tidy_heights
```
