```{r}
# 3. Shiny Server
server <- function(input, output) {
  
  output$mortality_plot <- renderPlot({
    # Filter df2 based on the user's sidebar selection
    filtered_data <- df2 %>% 
      filter(Entity %in% input$selected_countries)
    # Build the ggplot using the reactive filtered data
    ggplot(filtered_data, aes(x = Year, y = `Under.five.mortality.rate..selected.`, color = Entity, group = Entity)) +
      geom_line() +
      geom_point() + # Optional: adds points to the lines
      theme_minimal() +
      labs(y = "Under-five Mortality Rate", x = "Year")
  })
}
shinyApp(ui = ui, server = server)
```
