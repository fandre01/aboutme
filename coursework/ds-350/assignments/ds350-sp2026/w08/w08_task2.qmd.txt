### Filter the data
 
```{r}
# Filter wells_data
clean_wells <- wells_data %>%
  filter(Production > 5000)
```


```{r}
# Filter dams
clean_dams <- idaho_dams %>%
  filter(SurfaceAre > 50) 
```

```{r}
#Filter the water data
clean_water <- water_data %>%
  filter(FEAT_NAME == "Snake River" | FEAT_NAME == "Henrys Fork")
```

```{r}
# Filter State for just Idaho

clean_shp <- shp_data %>%
  filter(StateName == "Idaho")
```




```{r}
# let's Change the projection so that Idaho is not tilted when plotted

idaho_crs <- 6448
shp_proj <- st_transform(clean_shp, crs = idaho_crs)

wells_proj <- st_transform(clean_wells, crs = idaho_crs)

dams_proj <- st_transform(clean_dams, crs = idaho_crs)

water_proj <- st_transform(clean_water, crs = idaho_crs)



```





```{r}
# Let's make the graph

ggplot() +
  
  geom_sf(data = shp_proj, fill = "darkgreen", color = "white") +
  
  # Rivers layer 
  geom_sf(data = water_proj, color = "blue", size = 1) +
  
  # Wells and Dams 
  geom_sf(data = wells_proj, color = "yellow", size = 2) +
  geom_sf(data = dams_proj, color = "red", size = 2) +
   theme_bw()+
  labs(
    title = "Idaho Water Systems Map"
  )
```
