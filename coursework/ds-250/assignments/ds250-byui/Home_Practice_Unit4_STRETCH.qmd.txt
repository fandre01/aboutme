```{python}
import pandas as pd 
import numpy as np
```

```{python}
# import your data here using pandas and the URL

df = pd.read_csv("https://github.com/byuidatascience/data4dwellings/raw/master/data-raw/dwellings_ml/dwellings_ml.csv")
df.head()
```

```{python}
# create the feature and the target 
df["before1980"] = np.where(df["yrbuilt"] < 1980, 1, 0)
```

```{python}
# Select the feature and the target
y = df["before1980"]
x = df.select_dtypes(include="number").drop(columns=["before1980", "yrbuilt"])
```

```{python}
#split the model

x_train, x_test, y_train, y_test = train_test_split(
    x, y, test_size= 0.2, random_state= 42
)



```
```{python}
# build and fit the first model using Random State.

model = RandomForestClassifier(random_state= 2000)
model.fit(x_train, y_train)

```

```{python}
# prediction
pred = model.predict(x_test)
```

```{python}
# print acurancy

acc = accuracy_score(y_test, pred)
print("Accurancy:", acc)

print("\nConfusion Matrix")
print(confusion_matrix(y_test, pred))

print("\nClassification_report")
print(classification_report(y_test, pred))
```

```{python}
importances = pd.DataFrame({
    "feature": x_train.columns,
    "importance": model.feature_importances_
}).sort_values(by="importance", ascending=False).reset_index(drop=True)

importances

top10 = importances.head(10).iloc[::-1]


p1=(
    ggplot(top10, aes(x="feature", y="importance"))
    + geom_bar(stat="identity")
    + coord_flip()
    + ggtitle("Feature Importances")
)

p1
```
