```{python}
# Filter for name "John" and states Utah & Oregon
john_df = (
    df.query("name == 'John' and state in ['UT', 'OR']")
      .groupby(['year', 'state'], as_index=False)
      .agg(count=('count', 'sum'))
)

# Create the plot
p = (
    ggplot(john_df, aes(x='year', y='count', color='state'))
    + geom_line(size=1.2)
    
    # Vertical reference lines
    + geom_vline(xintercept=1936, linetype='dashed', color='black')
    + geom_vline(xintercept=1976, linetype='dashed', color='black')
    + geom_vline(xintercept=1999, linetype='dashed', color='black')

    # Labels
    + labs(
        title='The history of John for Utah (red) and Oregon (orange)',
        x='Year name given',
        y='Count of John'
    )

    # Manual color mapping
    + scale_color_manual(
        values={'UT': 'red', 'OR': 'orange'},
        name='State'
    )

    # Theme tweaks
    + theme_minimal()
    + theme(
        legend_position='right',
        plot_title=element_text(size=14, face='bold')
    )
)
```

```{python}
import pandas as pd
import numpy as np
from lets_plot import *
from sklearn.model_selection import train_test_split
from sklearn.ensemble import GradientBoostingClassifier
from sklearn.metrics import accuracy_score

LetsPlot.setup_html(isolated_frame=True)

```


```{python}
# import your data here using pandas and the URL
df = pd.read_csv("https://raw.githubusercontent.com/byuidatascience/data4dwellings/master/data-raw/dwellings_ml/dwellings_ml.csv")
df.head()
df.info()

```

```{python}
# Create target variable: has basement (1 = Yes, 0 = No)
df["has_basement"] = np.where(df["basement"] == "Yes", 1, 0)

df["has_basement"].value_counts()

```

```{python}
# Select features and target
y = df["has_basement"]

# Use numeric predictors only, drop target
X = df.select_dtypes(include="number").drop(columns=["has_basement"])

```

```{python}
# Train-test split (EXAM REQUIREMENTS)
X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.20,
    random_state=2020
)

X_train.shape, X_test.shape

```

```{python}
# Build and fit Gradient Boosting model
gb_model = GradientBoostingClassifier(random_state=2020)
gb_model.fit(X_train, y_train)

```

```{python}
# Make predictions
y_pred = gb_model.predict(X_test)

```

```{python}
# Evaluate accuracy
accuracy = accuracy_score(y_test, y_pred)
print("Model Accuracy:", accuracy)

```

```{python}
# Feature importance (Top 10)
feature_importance = pd.DataFrame({
    "feature": X.columns,
    "importance": gb_model.feature_importances_
}).sort_values(by="importance", ascending=False)

top10 = feature_importance.head(10)
top10

```

```{python}
# Feature importance plot (Top 10)
(
    ggplot(top10, aes(x="reorder(feature, importance)", y="importance"))
    + geom_col(fill="#4C72B0")
    + coord_flip()
    + labs(
        title="Top 10 Most Important Features for Predicting a Basement",
        x="Feature",
        y="Importance"
    )
    + theme_minimal()
)

```
