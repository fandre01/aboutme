```{python}
summary = df.groupby("airport_code").agg(
    total_flights =("num_of_flights_total", "sum"),
    weather=("num_of_delays_weather", "sum"),
    carrier=("num_of_delays_carrier", "sum"),
    security=("num_of_delays_security", "sum")
).reset_index()
summary
```

```{python}
summary["prop_weather"] = summary["weather"] / summary["total_flights"]
summary["prop_carrier"] = summary["carrier"] / summary["total_flights"]
summary["prop_security"] = summary["security"] / summary["total_flights"]
```

```{python}
plot_df = summary.melt(
  id_vars="airport_code",
  value_vars=["prop_weather", "prop_carrier", "prop_security"],
  var_name="delay_type",
  value_name="proportion"
)
plot_df
```
```{python}
(
  ggplot(plot_df, aes("delay_type", "proportion", fill="delay_type"))
```
