```{python}
john_long = (
    df[df['name'] == 'John'][['year', 'UT', 'OR']]
    .melt(
        id_vars='year',
        value_vars=['UT', 'OR'],
        var_name='state',
        value_name='count'
    )
)

from lets_plot import element_line

p = (
    ggplot(john_long, aes(x='year', y='count', color='state'))
    + geom_line(size=1.2)
    + geom_vline(xintercept=1936, color = "black")
    + geom_vline(xintercept=1976, color = "black")
    + geom_vline(xintercept=1999, color = "black")
    + scale_x_continuous(limits=[1910, 2020], format="d")
    + labs(
        title='The history of John for Utah and Oregon',
        x='Year name given',
        y='Count of John'
    )
    + scale_color_manual(
        values={'UT': 'red', 'OR': 'orange'},
        name='State'
    )
    + theme_minimal()
    + theme(
        axis_line_x=element_line(color='black', size=1),
        axis_line_y=element_line(color='black', size=1)
    )
        # Label box
    + geom_text(
        x=1930,
        y=400,
        label='1936',
       color = "black"
       
    )
    + geom_text(
        x=1970,
        y=700,
        label='1976',
       color = "black"
       
    )
    + geom_text(
        x=1993,
        y=300,
        label='1999',
       color = "black"
       
    )
)

p
```
