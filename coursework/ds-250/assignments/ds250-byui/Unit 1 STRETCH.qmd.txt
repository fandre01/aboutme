# Q1
elliot = df.query('name== "Elliot"')

Chart = (
    ggplot(data= elliot, mapping= aes(x = "year", y = "Total", color = 'name'))
    + scale_color_manual(values={'Elliot': 'blue'})
    + geom_line()
    + geom_vline(xintercept=1982, linetype='dashed', color="red")
    + scale_x_continuous(limits=[1950, 2020], format='d')
    + geom_text(x=1975, y=1250, label="E.T Released", color='black', size= 5)
    + geom_vline(xintercept=1985, linetype='dashed', color="red")
    + geom_text(x=1992, y=1250, label="Second Release", color='black', size= 5)
    + geom_vline(xintercept=2003, linetype='dashed', color="red")
    + geom_text(x=2009, y=1250, label="Third Release", color="black", size= 5)

    +theme(
        panel_grid=element_line(color = "white"),
        panel_background= element_rect(fill="#E5ECF6")
    )
    
)
Chart
