```{python}
# --- Conditional filtering ---
# Which rows are for Japan?
df[df['Entity'] == 'Japan']
# Which rows are for France, Japan, or El Salvador?
df[df['Entity'].isin(['France', 'Japan', 'El Salvador'])]
# Using query syntax
df.query('Entity == "Japan"')

df.query('Entity == ["France", "Japan", "El Salvador"]')
 # everything except France
df.query('Entity != "France"')
```
