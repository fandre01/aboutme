```{python}
# Include and execute your code here
# Create the the model.

df["before1980"] = np.where(df["yrbuilt"] < 1980, 1, 0)

```

```{python}
#select the feature and Target

y = df["before1980"]

x = df.select_dtypes(include="number").drop(columns=["before1980", "yrbuilt"])
```

```{python}
# Split the model

x_train, x_test, y_train, y_test = train_test_split(
  x,y,test_size=0.2, random_state= 42
)
```

```{python}
# build the model and fit the model

model = RandomForestClassifier(random_state=42)
model.fit(x_train, y_train)

```

```{python}
# make prediction
pred = model.predict(x_test)
```

```{python}
# Evaluate the model

acc = accuracy_score(y_test, pred)
print("Accuracy:", acc)

print("\nprecision_score")
print(precision_score)

pre = precision_score(y_test, pred)
print("\nprecision:",pre)

recall = recall_score(y_test, pred)
print("\nRecall:", recall)

f1 = f1_score(y_test, pred)
print("F1_Score:", f1)

ar = roc_auc_score(y_test, pred)
print("Area:", ar)

print("\nClassification Report")
print(classification_report(y_test, pred))
```
