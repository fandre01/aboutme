# Step 1: Count how many have seen each episode
seen_summary = (
    df_seen[seen_cols]
    .notna()
    .sum()
    .reset_index()
)
seen_summary.columns = ["Film", "Count"]

# Step 2: Fix film names
seen_summary["Film"] = seen_summary["Film"].str.replace("seen_epi", "Episode ")


# Step 3: Add percentages
total = len(df_seen)
seen_summary["Percentage"] = round(seen_summary["Count"] / total * 100, 1)
seen_summary["PctLabel"] = seen_summary["Percentage"].astype(str) + "%"

# 1. Melt data
df_long = df_seen.melt(
    id_vars=None,
    value_vars=[f"rank_epi{i}" for i in range(1, 7)],
    var_name="Film",
    value_name="Rank"
)

# 2. Clean film names
df_long["Film"] = df_long["Film"].str.replace("rank_epi", "Episode ")

# 3. Rank category
def bucket(r):
    if r <= 2: return "Top third"
    if r <= 4: return "Middle third"
    return "Bottom third"

df_long["Category"] = df_long["Rank"].apply(bucket)

# 4. Summary
df_summary = (
    df_long.groupby(["Film", "Category"])
           .size()
           .reset_index(name="Count")
)

df_summary["Percentage"] = df_summary["Count"] / len(df_seen) * 100
df_summary["PctLabel"] = df_summary["Percentage"].round(1).astype(str) + "%"

# Third Graph
plot = (
    ggplot(df_summary, aes("Film", "Percentage", fill="Category"))
    + geom_bar(stat="identity")
    + facet_grid("Category")
    + geom_text(aes(label="PctLabel"), nudge_y=2)
    + coord_flip()
    + labs(
        title = "How People Rate the 'Star Wars' Movies",
        subtitle = "How often each film was rated in the top, middle"'\n'
          "and bottom third (by 471 respondents who have seen all six films)" '\n'
        )
        + theme(
        plot_title=element_text(face="bold"),
        )
      
)
