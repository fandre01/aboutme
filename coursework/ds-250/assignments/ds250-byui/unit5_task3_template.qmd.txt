def income_mapping(range):
  if range == '$0 - $24,999':
    return 12500
  elif range == '$25, 000 - $49,999':
    return 37500
  elif range == '$100,000 - $149,999':
    return 125000
  elif range == '$50, 000 - $99,999':
    return 75000
  elif range == '$150,000+':
    return 150000
  else:
    np.nan


df_cleaned['household_income'] = df_cleaned['income'].apply(income_mapping).astype('float64')
